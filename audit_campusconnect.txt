services:
  postgres-auth:
    image: postgres:16-alpine
    container_name: campus-postgres-auth
    environment:
      POSTGRES_DB: ${AUTH_DB_NAME:-auth_db}
      POSTGRES_USER: ${AUTH_DB_USER:-campus_auth}
      POSTGRES_PASSWORD: ${AUTH_DB_PASSWORD:-auth_password_dev}
    ports:
      - "${POSTGRES_AUTH_PORT:-5433}:5432"
    volumes:
      - pg_auth_data:/var/lib/postgresql/data
    networks:
      - campus-net
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ${AUTH_DB_USER:-campus_auth} -d ${AUTH_DB_NAME:-auth_db}"]
      interval: 10s
      timeout: 5s
      retries: 5

  postgres-student:
    image: postgres:16-alpine
    container_name: campus-postgres-student
    environment:
      POSTGRES_DB: ${STUDENT_DB_NAME:-student_db}
      POSTGRES_USER: ${STUDENT_DB_USER:-campus_student}
      POSTGRES_PASSWORD: ${STUDENT_DB_PASSWORD:-student_password_dev}
    ports:
      - "${POSTGRES_STUDENT_PORT:-5434}:5432"
    volumes:
      - pg_student_data:/var/lib/postgresql/data
    networks:
      - campus-net
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ${STUDENT_DB_USER:-campus_student} -d ${STUDENT_DB_NAME:-student_db}"]
      interval: 10s
      timeout: 5s
      retries: 5

  kafka:
    image: apache/kafka:3.8.1
    container_name: campus-kafka
    ports:
      - "${KAFKA_PORT:-9092}:9092"
    environment:
      KAFKA_NODE_ID: 1
      KAFKA_PROCESS_ROLES: broker,controller
      KAFKA_LISTENERS: PLAINTEXT://0.0.0.0:9092,CONTROLLER://0.0.0.0:9093
      KAFKA_ADVERTISED_LISTENERS: PLAINTEXT://${KAFKA_ADVERTISED_HOST:-kafka}:${KAFKA_ADVERTISED_PORT:-9092}
      KAFKA_CONTROLLER_LISTENER_NAMES: CONTROLLER
      KAFKA_LISTENER_SECURITY_PROTOCOL_MAP: CONTROLLER:PLAINTEXT,PLAINTEXT:PLAINTEXT
      KAFKA_CONTROLLER_QUORUM_VOTERS: 1@kafka:9093
      KAFKA_OFFSETS_TOPIC_REPLICATION_FACTOR: 1
      KAFKA_TRANSACTION_STATE_LOG_REPLICATION_FACTOR: 1
      KAFKA_TRANSACTION_STATE_LOG_MIN_ISR: 1
      CLUSTER_ID: ${KAFKA_CLUSTER_ID:-MkU3OEVBNTcwNTJENDM2Qk}
    volumes:
      - kafka_data:/var/lib/kafka/data
    networks:
      - campus-net
    healthcheck:
      test: ["CMD-SHELL", "/opt/kafka/bin/kafka-topics.sh --bootstrap-server localhost:9092 --list > /dev/null 2>&1"]
      interval: 15s
      timeout: 10s
      retries: 5

  eureka-server:
    build:
      context: ./eureka-server
    container_name: campus-eureka-server
    ports:
      - "${EUREKA_SERVER_PORT:-8761}:8761"
    networks:
      - campus-net

  config-server:
    build:
      context: ./config-server
    container_name: campus-config-server
    ports:
      - "${CONFIG_SERVER_PORT:-8888}:8888"
    environment:
      SPRING_CLOUD_CONFIG_SERVER_GIT_URI: ${CONFIG_GIT_URI:-file:///config-repo}
      SPRING_CLOUD_CONFIG_SERVER_GIT_DEFAULT_LABEL: ${CONFIG_GIT_DEFAULT_LABEL:-main}
    volumes:
      - ./config-repo:/config-repo
    networks:
      - campus-net

networks:
  campus-net:
    driver: bridge
